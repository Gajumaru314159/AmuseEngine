360 template<class T>
361 void* ServiceBuilder<T>::create(ServiceContainer& container) {
362 // 生成
363 Internal::ServiceHolder<T>* holder = nullptr;
364 if (m_getter) {
365 holder = new Internal::ServiceHolder<T>(m_getter(), false);
366 }
367 else {
368 holder = new Internal::ServiceHolder<T>(reinterpret_cast<T*>(Internal::Factory<T>::Create(m_injector, container)), true);
369 }
370 // 無効
371 if (holder->get() == nullptr) {
372 delete holder;
373 return nullptr;
374 }
375 // 基底クラスをマッピング
376 auto index = container.m_services.size();
377 for (auto& type : m_bases) {
378 container.m_indices.emplace(type, index);
379 }
380 return container.m_services.emplace_back(holder)->get();
381 }

384 template<class T>
385 T* ServiceInjector::create(ServiceContainer& container)const {
386 // 生成済み
387 if (auto instance = container.get<T>())
388 return instance;
389 // 抽象->具象
390 Vector<Type> fallback;
391 const auto& concretes = try_find(m_builderMap, Type::Get<T>(), fallback);
392 // 生成
393 for (auto& concrete : concretes) {
394 auto& builder = m_builders.find(concrete)->second;
395 try {
396 return reinterpret_cast<T*>(builder->create(container));
397 }
398 catch (const std::exception& e) {
399 LOG_TRACE("[DI] {}の生成がキャンセルされました。\n{}", Type::Get<T>().name(), e.what());
400 }
401 }
402 return nullptr;
403 }
