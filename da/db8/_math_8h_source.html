255 // @brief 高速余弦正弦関数
256 inline void Math::SinCos(f32 radians, f32& outSin, f32& outCos)noexcept {
257 radians = Math::Wrap(radians, -PI, PI);
258
259 static f32 waru[8] = { -1.0f / (3 * 4 * 5 * 6 * 7 * 8),-1.0f / (2 * 3 * 4 * 5 * 6 * 7),1.0f / (3 * 4 * 5 * 6),1.0f / (2 * 3 * 4 * 5),-1.0f / (3 * 4),-1.0f / (2 * 3),1.0f,1.0f };
260 f32 c = 1.0f / (3 * 4 * 5 * 6 * 7 * 8 * 9 * 10);
261 f32 s = 1.0f / (2 * 3 * 4 * 5 * 6 * 7 * 8 * 9);
262 f32* p = waru;
263 f32 z;
264
265 s32 i;
266 radians *= (1.0f / 32.0f);
267 z = radians * radians;
268 do {
269 c = c * z + (*p);
270 p++;
271 s = s * z + (*p);
272 p++;
273 } while (p < waru + 8);
274 c = c * z;
275 s = s * radians;
276 for (i = 0; i < 5; i++) {
277 s = s * (2.0f - c);
278 c = c * (4.0f - c);
279 }
280 outSin = s;
281 outCos = 1.0f - c / 2.0f;
282 }
