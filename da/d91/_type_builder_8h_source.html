506 template< class M, class... Args, class... Names>
507 auto _method_impl(StringView name, M method, Names&&... argNames)
508 -> std::enable_if_t<MethodTraits<M>::Count == sizeof...(Names) || 0 == sizeof...(Names), TagBuilder >
509 {
510 AMUSE_ASSERT(!m_info.methods.contains(name), "{}は登録済みのメソッドです [{}]", name, m_info.type.name());
511
512 m_info.methodOrder.emplace_back(name);
513 auto& info = m_info.methods[name];
514 info.name = name;
515 info.returnType = Type::Get<typename MethodTraits<M>::return_type>();
516 info.isConst = MethodTraits<M>::Const;
517
518 // 0引数(引数名未指定)に対応するために最後尾に空要素を追加している
519 Array<StringView,sizeof...(Args)+1> names = {StringView(argNames)... ,""};
520 auto types = MethodTraits<M>::Types();
521
522 for (s32 i = 0; i + 1 < std::size(types); ++i) {
523 auto& arg = info.arguments.emplace_back();
524 arg.type = types.at(i);
525 arg.name = (sizeof...(Names) == 0) ? _GetDefaultArgumentName(i) : names.at(i);
526 }
527
528 if constexpr (sizeof...(Args) == 0) {
529 info.invoke = [=](Any& owner, Span<Any> args) { return _InvokeWithoutArgs<M, Args...>(owner, args, method); };
530 }
531 else {
532 info.invoke = [=](Any& owner, Span<Any> args) { return _InvokeMethod<M, Args...>(owner, args, method); };
533 }
534
535 return info;
536 }

546 m_info.propertyOrder.emplace_back(name);

559 }
560 };

562 info.setter = [=](Any& owner, const Any& value) {
563 (owner.as<T>().*(address)) = value.as<TField>();
564 if constexpr (!std::is_same_v<FCallback, std::nullptr_t>) {
565 (owner.as<T>().*callback)();
566 }
567 };
568 } else {
569 static_assert(std::is_same_v<FCallback, std::nullptr_t>, "const field cannot have on-edit callback because it is not writable");
570 }
