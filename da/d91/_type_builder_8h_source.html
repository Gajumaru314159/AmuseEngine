522 auto _method_impl(StringView name, M method, Names&&... argNames)
523 -> std::enable_if_t<MethodTraits<M>::Count == sizeof...(Names) || 0 == sizeof...(Names), TagBuilder >
524 {
525 AMUSE_ASSERT(!m_info.methods.contains(name), "{}は登録済みのメソッドです [{}]", name, m_info.type.name());

527 m_info.methodOrder.emplace_back(name);
528 auto& info = m_info.methods[name];
529 info.name = name;
530 info.returnType = Type::Get<typename MethodTraits<M>::return_type>();
531 info.isConst = MethodTraits<M>::Const;
532
533 // 0引数(引数名未指定)に対応するために最後尾に空要素を追加している
534 Array<StringView,sizeof...(Args)+1> names = {StringView(argNames)... ,""};
535 auto types = MethodTraits<M>::Types();
536
537 for (s32 i = 0; i + 1 < std::size(types); ++i) {
538 auto& arg = info.arguments.emplace_back();
539 arg.type = types.at(i);
540 arg.name = (sizeof...(Names) == 0) ? _GetDefaultArgumentName(i) : names.at(i);
541 }
542
543 if constexpr (sizeof...(Args) == 0) {
544 info.invoke = [=](Any& owner, Span<Any> args) { return _InvokeWithoutArgs<M, Args...>(owner, args, method); };
545 }
546 else {
547 info.invoke = [=](Any& owner, Span<Any> args) { return _InvokeMethod<M, Args...>(owner, args, method); };
548 }

580 (owner.as<T>().*callback)();
581 }
582 };
583 } else {
