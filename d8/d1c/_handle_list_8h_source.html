739 inline void HandleList<T>::insert_impl(Handle* pPrev, Handle& h) {
740 assert(pPrev != nullptr);
741 assert(has(*pPrev));// 違うコンテナのイテレータならエラー

744 h.pParent = this;

749

751 pPrev->pPrev = &h;

755 h.pNext = pPrev;
