642 if (first == last) {
643 return iterator(this, first.m_index);

646 size_type start = first.m_index;

716 value_type remove_swap_last(size_type index) {
717 AMUSE_ASSERT_EXPR(index < size());
718 value_type removed(m_keys[index], m_values[index]);
719 const size_type lastIndex = size() - 1;
720 key_type keyCopy = m_keys[index];
721
722 m_index.erase(keyCopy);
723
724 if (index != lastIndex) {
725 m_keys[index] = std::move(m_keys[lastIndex]);
726 m_values[index] = std::move(m_values[lastIndex]);
727 m_index[m_keys[index]] = index;
728 }

731 m_values.pop_back();
