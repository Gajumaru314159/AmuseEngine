103 class ScheduleExecutor {
104 public:
107 explicit ScheduleExecutor(const ScheduleExecutorConfig& config = {}) {
108 s32 count = config.workerCount;
109 if (count <= 0) {
110 count = static_cast<s32>(std::thread::hardware_concurrency());
111 if (count <= 0) count = 1;
112 }
113 m_running = true;
114 for (s32 i = 0; i < count; ++i) {
115 m_workers.emplace_back([this] { process(); });
116 }
117 }
118
121 ~ScheduleExecutor() {
122 wait();
123 {
124 std::lock_guard lock(m_mutex);
125 m_running = false;
126 }
127 m_condition.notify_all();
128 for (auto& worker : m_workers) {
129 if (worker.joinable()) worker.join();
130 }
131 }

133 ScheduleExecutor(const ScheduleExecutor&) = delete;
134 ScheduleExecutor& operator=(const ScheduleExecutor&) = delete;
135
137 auto workerCount() const -> s32 { return static_cast<s32>(m_workers.size()); }

141 void enqueue(Action&& action) {
142 {
143 std::lock_guard lock(m_mutex);
144 ++m_pendingCount;
145 m_queue.emplace(std::move(action));
146 }
147 m_condition.notify_one();
148 }
149
151 void wait() {
152 std::unique_lock lock(m_mutex);
153 m_doneCondition.wait(lock, [this] { return m_pendingCount == 0 && m_queue.empty(); });
154 }
155
157 auto isWorkerThread() const -> bool { return t_currentExecutor == this; }

159 private:
161 void process() {
162 t_currentExecutor = this;
163 while (true) {
164 Action action;
165 {
166 std::unique_lock lock(m_mutex);
167 m_condition.wait(lock, [this] { return !m_queue.empty() || !m_running; });
168 if (!m_running && m_queue.empty()) break;
169 action = std::move(m_queue.front());
170 m_queue.pop();
171 }
172 try {
173 action();
174 } catch (...) {
175 std::lock_guard lock(m_mutex);
176 if (!m_exception) m_exception = std::current_exception();
177 }
178 {
179 std::lock_guard lock(m_mutex);
180 --m_pendingCount;
181 if (m_pendingCount == 0 && m_queue.empty()) m_doneCondition.notify_all();
182 }
183 }
184 t_currentExecutor = nullptr;
185 }

190 std::condition_variable m_condition;
191 std::condition_variable m_doneCondition;
192 bool m_running = false;
193 s32 m_pendingCount = 0;
194 std::queue<Action> m_queue;
195 Vector<std::thread> m_workers;
196 std::exception_ptr m_exception;

784 void expandSetHierarchy(ScheduleBuildResult& result) {
785 const size_t setCount = m_sets.size();
786 Vector<Vector<s32>> children(setCount);
787 for (const auto& set : m_sets) {
788 for (s32 parent : set.parents) {
789 if (parent == set.id) {
790 result.addError({ ScheduleBuildErrorType::InvalidSetHierarchy, "set cannot contain itself", { set.name } });
791 } else if (parent < 0 || static_cast<size_t>(parent) >= setCount) {
792 result.addError({ ScheduleBuildErrorType::InvalidSetHierarchy, "invalid parent set", { set.name } });
793 } else {
794 children[static_cast<size_t>(parent)].emplace_back(set.id);
795 }
796 }
797 }
798 if (!result) return;
799
800 Vector<s32> state(setCount, 0);
801 for (const auto& set : m_sets) {
802 if (hasSetCycle(set.id, children, state)) {
803 result.addError({ ScheduleBuildErrorType::InvalidSetHierarchy, "set hierarchy has cycle", { set.name } });
804 return;
805 }
806 }
807
808 for (auto& system : m_systems) {
809 Vector<s32> expanded = system.sets;
810 for (s32 setId : system.sets) {
811 collectParentSets(setId, expanded);
812 }
813 std::sort(expanded.begin(), expanded.end());
814 expanded.erase(std::unique(expanded.begin(), expanded.end()), expanded.end());
815 system.sets = std::move(expanded);
816 }
817 }

820 bool hasSetCycle(s32 setId, const Vector<Vector<s32>>& children, Vector<s32>& state) const {
821 auto index = static_cast<size_t>(setId);
822 if (state[index] == 1) return true;
823 if (state[index] == 2) return false;
824 state[index] = 1;
825 for (s32 child : children[index]) {
826 if (hasSetCycle(child, children, state)) return true;
827 }
828 state[index] = 2;
829 return false;
830 }

841 void expandEdges(ScheduleBuildResult& result) {
842 std::set<std::pair<s32, s32>> edges;
843 for (const auto& edge : m_explicitEdges) {
844 Vector<s32> fromSystems = resolveRef(edge.from);
845 Vector<s32> toSystems = resolveRef(edge.to);
846 if (fromSystems.empty() || toSystems.empty()) continue;
847 for (s32 from : fromSystems) {
848 for (s32 to : toSystems) {
849 if (from == to) continue;
850 edges.emplace(from, to);
851 }
852 }
853 }
854
855 for (const auto& set : m_sets) {
856 if (!set.chain) continue;
857 Vector<s32> systemsInSet = resolveRef({ Internal::ScheduleNodeKind::Set, set.id });
858 std::sort(systemsInSet.begin(), systemsInSet.end());
859 for (size_t i = 1; i < systemsInSet.size(); ++i) {
860 edges.emplace(systemsInSet[i - 1], systemsInSet[i]);
861 }
862 }
863
864 for (const auto& [from, to] : edges) {
865 m_compiledEdges.emplace_back(
866 Internal::ScheduleNodeRef{ Internal::ScheduleNodeKind::Task, from },
867 Internal::ScheduleNodeRef{ Internal::ScheduleNodeKind::Task, to }
868 );
869 }
870 (void)result;
871 }

874 auto resolveRef(Internal::ScheduleNodeRef ref) const -> Vector<s32> {
875 Vector<s32> systems;
876 if (ref.kind == Internal::ScheduleNodeKind::Task) {
877 if (isValidRef(ref)) systems.emplace_back(ref.id);
878 return systems;
879 }
880 for (const auto& system : m_systems) {
881 if (std::find(system.sets.begin(), system.sets.end(), ref.id) != system.sets.end()) {
882 systems.emplace_back(system.id);
883 }
884 }
885 return systems;
886 }

889 void buildExecutionPlan(ScheduleBuildResult& result) {
890 const size_t count = m_systems.size();
891 Vector<Vector<s32>> outgoing(count);
892 Vector<s32> indegree(count, 0);
893 for (const auto& edge : m_compiledEdges) {
894 const s32 from = edge.from.id;
895 const s32 to = edge.to.id;
896 outgoing[static_cast<size_t>(from)].emplace_back(to);
897 ++indegree[static_cast<size_t>(to)];
898 }
899
900 Vector<s32> ready;
901 for (const auto& system : m_systems) {
902 if (indegree[static_cast<size_t>(system.id)] == 0) ready.emplace_back(system.id);
903 }
904
905 Vector<s32> indegreeForBatch = indegree;
906 while (!ready.empty()) {
907 if (m_buildSettings.stableTopologicalSort) std::sort(ready.begin(), ready.end());
908
909 Vector<s32> batchSystems;
910 Vector<s32> deferred;
911 for (s32 id : ready) {
912 if (canAddToBatch(batchSystems, id)) {
913 batchSystems.emplace_back(id);
914 } else {
915 deferred.emplace_back(id);
916 }
917 }
918 if (batchSystems.empty()) {
919 batchSystems.emplace_back(deferred.front());
920 deferred.erase(deferred.begin());
921 }
922
923 ExecutionBatch batch;
924 batch.systems = batchSystems;
925 m_batches.emplace_back(std::move(batch));
926 for (s32 id : batchSystems) m_order.emplace_back(id);
927
928 Vector<s32> next = deferred;
929 for (s32 id : batchSystems) {
930 for (s32 to : outgoing[static_cast<size_t>(id)]) {
931 auto& value = indegreeForBatch[static_cast<size_t>(to)];
932 --value;
933 if (value == 0) next.emplace_back(to);
934 }
935 }
936 ready = std::move(next);
937 }
938
939 if (m_order.size() != count) {
940 Vector<String> nodes;
941 for (const auto& system : m_systems) {
942 if (std::find(m_order.begin(), m_order.end(), system.id) == m_order.end()) {
943 nodes.emplace_back(system.name);
944 }
945 }
946 result.addError({ ScheduleBuildErrorType::DependencyCycle, "schedule dependency cycle detected", nodes });
947 }
948 }

951 bool canAddToBatch(const Vector<s32>& batch, s32 id) const {
952 const auto& target = m_systems[static_cast<size_t>(id)];
953 for (s32 otherId : batch) {
954 const auto& other = m_systems[static_cast<size_t>(otherId)];
955 if (target.exclusive || other.exclusive) return false;
956 for (const auto& tag : target.conflicts) {
957 if (std::find(other.conflicts.begin(), other.conflicts.end(), tag) != other.conflicts.end()) return false;
958 }
959 }
960 return true;
961 }

971 void runMultiThreaded(Args... args) {
972 ScheduleExecutor localExecutor;
973 ScheduleExecutor& executor = m_executor ? *m_executor : localExecutor;
974 std::exception_ptr firstException;
975 std::mutex exceptionMutex;
976
977 for (const auto& batch : m_batches) {
978 for (s32 id : batch.systems) {
979 auto& system = m_systems[static_cast<size_t>(id)];
980 if (system.mainThread) continue;
981 executor.enqueue([&system, &args..., &firstException, &exceptionMutex] {
982 try {
983 system.invoker(args...);
984 } catch (...) {
985 std::lock_guard lock(exceptionMutex);
986 if (!firstException) firstException = std::current_exception();
987 }
988 });
989 }
990
991 for (s32 id : batch.systems) {
992 auto& system = m_systems[static_cast<size_t>(id)];
993 if (!system.mainThread) continue;
994 try {
995 system.invoker(args...);
996 } catch (...) {
997 std::lock_guard lock(exceptionMutex);
998 if (!firstException) firstException = std::current_exception();
999 }
1000 }
1001
1002 executor.wait();
1003 if (firstException) std::rethrow_exception(firstException);
1004 }
