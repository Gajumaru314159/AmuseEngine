801 inline Vec4 Vec4::Slerp(const Vec4& a, const Vec4& b, f32 t) {
802 f32 al, bl;
803 Vec4 an, bn;
804 a.toDirectionAndLength(an, al);
805 b.toDirectionAndLength(bn, bl);
806 f32 angle = Math::Acos(Dot(an, bn));
807 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
808 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);
809
810 return Vec4(
811 an.x * ps + bn.x * pe,
812 an.y * ps + bn.y * pe,
813 an.z * ps + bn.z * pe,
814 an.w * ps + bn.w * pe
815 );
816 }
