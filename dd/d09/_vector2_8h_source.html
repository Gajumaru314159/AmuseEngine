787 inline Vec2 Vec2::Slerp(const Vec2& a, const Vec2& b, f32 t) {
788 f32 al, bl;
789 Vec2 an, bn;
790 a.toDirectionAndLength(an, al);
791 b.toDirectionAndLength(bn, bl);
792 f32 angle = Math::Acos(Dot(an, bn));
793 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
794 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);
795
796 return Vec2(
797 an.x * ps + bn.x * pe,
798 an.y * ps + bn.y * pe
799 );
800 }
