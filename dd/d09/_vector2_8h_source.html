788 inline Vec2 Vec2::Slerp(const Vec2& a, const Vec2& b, f32 t) {
789 f32 al, bl;
790 Vec2 an, bn;
791 a.toDirectionAndLength(an, al);
792 b.toDirectionAndLength(bn, bl);
793 f32 angle = Math::Acos(Dot(an, bn));
794 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
795 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);
796
797 return Vec2(
798 an.x * ps + bn.x * pe,
799 an.y * ps + bn.y * pe
800 );
801 }
