211 constexpr u32 IntColor::toCode(ColorCodeFormat format)const noexcept {
212 switch (format) {
213 case ColorCodeFormat::ARGB:
214 return (a << 24) | (r << 16) | (g << 8) | (b);
215 case ColorCodeFormat::ABGR:
216 return (a << 24) | (b << 16) | (g << 8) | (r);
217 case ColorCodeFormat::RGBA:
218 return (r << 24) | (g << 16) | (b << 8) | (a);
219 case ColorCodeFormat::BGRA:
220 return (b << 24) | (g << 16) | (r << 8) | (a);
221 }
222 return 0xFFFFFFFF;
223 }
