210 constexpr u32 IntColor::toCode(ColorCodeFormat format)const noexcept {
211 switch (format) {
212 case ColorCodeFormat::ARGB:
213 return (a << 24) | (r << 16) | (g << 8) | (b);
214 case ColorCodeFormat::ABGR:
215 return (a << 24) | (b << 16) | (g << 8) | (r);
216 case ColorCodeFormat::RGBA:
217 return (r << 24) | (g << 16) | (b << 8) | (a);
218 case ColorCodeFormat::BGRA:
219 return (b << 24) | (g << 16) | (r << 8) | (a);
220 }
221 return 0xFFFFFFFF;
222 }
