214 SpinLock m_lock;

219 public:
220
222 SyncHandleManager(size_t capacity)

225
226 }
227
228 ~SyncHandleManager() {
229 for (auto& handle : m_actives) {
230 remove(handle);
231 }
232 update();
233 }
234
236 handle_type add(T* instance) {
237 ScopeLock lock(m_lock);
238 auto handle = m_manager.add(instance);
239 m_actives.emplace(handle);
240 return handle;
241 }
242
244 void remove(const handle_type& handle) {
245 if (handle.empty())return;
246 ScopeLock lock(m_lock);
247 if (m_disposedSet.contains(handle.id()))return;
248 m_disposedSet.emplace(handle.id());
249 m_disposeQueue.push_back(handle);
250 }
251
252 T* at(const handle_type& handle)const {
253 return m_manager.at(handle);
254 }
255
257 void visit(Func<void(T&)>visitor) {
258 ScopeLock lock(m_lock);
259 for (auto& handle : m_actives) {
260 if (auto instance = handle.value()) {
261 visitor(*instance);
262 }
263 }
264 }
265
267 void update() {
268 ScopeLock lock(m_lock);
269 for (auto& handle : m_disposeQueue) {
270 m_actives.erase(handle);
271 m_manager.remove(handle);
272 }
273 m_disposeQueue.clear();
274 m_disposedSet.clear();
275 }
