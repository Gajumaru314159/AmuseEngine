159 inline Vec3 Triangle::innerGravity()const noexcept {
160 f32 a = Vec3::Dist(p1, p2);
161 f32 b = Vec3::Dist(p2, p0);
162 f32 c = Vec3::Dist(p0, p1);
163 if (a + b + c < Math::EPSILON)return p0;
164 return (p0 * a + p1 * b + p2 * c) / (a + b + c);
165 }

169 inline Vec3 Triangle::circumcenter()const noexcept {
170 auto a = Vec3::SqrDist(p1, p2);
171 auto b = Vec3::SqrDist(p2, p0);
172 auto c = Vec3::SqrDist(p0, p1);
173 auto t_a = a * (-a + b + c);
174 auto t_b = b * (a - b + c);
175 auto t_c = c * (a + b - c);
176 auto s = t_a + t_b + t_c;
177 return t_a / s * p0 + t_b / s*p1 +t_c / s * p2;
178 }

182 inline f32 Triangle::area()const noexcept {

185 }
