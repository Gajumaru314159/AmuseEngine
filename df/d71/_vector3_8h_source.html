1040 inline Vec3 Vec3::Slerp(const Vec3& a, const Vec3& b, f32 t) {
1041 f32 al, bl;
1042 Vec3 an, bn;
1043 a.toDirectionAndLength(an, al);
1044 b.toDirectionAndLength(bn, bl);
1045 f32 angle = Math::Acos(Dot(an, bn));
1046 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
1047 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);
1048
1049 return Vec3(
1050 an.x * ps + bn.x * pe,
1051 an.y * ps + bn.y * pe,
1052 an.z * ps + bn.z * pe
1053 );
