1041 inline Vec3 Vec3::Slerp(const Vec3& a, const Vec3& b, f32 t) {
1042 f32 al, bl;
1043 Vec3 an, bn;
1044 a.toDirectionAndLength(an, al);
1045 b.toDirectionAndLength(bn, bl);
1046 f32 angle = Math::Acos(Dot(an, bn));
1047 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
1048 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);

1050 return Vec3(
1051 an.x * ps + bn.x * pe,
1052 an.y * ps + bn.y * pe,
