297 constexpr Quat& Quat::operator *= (const Quat& quat) noexcept {
298 f32 x = this->x;
299 f32 y = this->y;
300 f32 z = this->z;
301 f32 w = this->w;
302
303 this->x = y * quat.z - z * quat.y + x * quat.w + w * quat.x;
304 this->y = z * quat.x - x * quat.z + y * quat.w + w * quat.y;
305 this->z = x * quat.y - y * quat.x + z * quat.w + w * quat.z;
306 this->w = w * quat.w - x * quat.x - y * quat.y - z * quat.z;
307 return *this;
308 }
